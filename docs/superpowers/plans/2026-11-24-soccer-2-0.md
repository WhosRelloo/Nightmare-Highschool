# Soccer 2.0 (the FIER clips in Soccer 1.1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** every soccer move with a FIER clip plays it, the foot meets the ball on the contact frame, and in a soccer
match the FIER jogs and idles replace the default walk and idle, with no server change and the 1.1 distances kept.

**Architecture:**
- **A bake** (Edit-mode Luau) copies the 24 clips from `Workspace."Soccer Anims".AnimSaves` into
  `ReplicatedStorage.NHSoccerClips`: baked travel removed, `Contact` markers, a right-foot mirror of the kick, loop
  flags. It also writes the generated `ReplicatedStorage.Config.SoccerClips` (lengths, contacts, travel paths, strides).
- **A pure module** `ReplicatedStorage.Functions.SoccerAnim` does the timing maths (kick start, charge creep, travel
  window, rate fit, path progress, slot picks). It's tested in a VM.
- **A client module** `Controllers.Gui.Soccer.Clips` resolves a slot to an animation (published id, else the
  registered Studio copy, else the 1.1 clip). It plays a clip through its travel window and runs the travel dash.
  `Controllers.Gui.Soccer` routes every server Fx through it.
- **A client controller** `Controllers.Character.SoccerPack` swaps the movement pack during a soccer match. It uses a
  new `AnimationController.Override` for the jog stride, a real backpedal track and the keeper's strafes.

**Tech Stack:** Luau in Roblox Studio (NH Combat Test), KeyframeSequenceProvider, the repo harness (`rbxc.py`,
`tools/admin-console/run_check.py`, `run_live.py`, `install_new.py`, `install_inplace.py`, `shot.py`,
`tools/forms/harness/round_start.sh`, `restore.py`), Python 3 for drivers and audits.

```lua
--!strict
--[[ (soccer-2.0) Pure soccer-clip maths: no services, no instances. A Clip is a Config.SoccerClips.Clips entry:
	{ Length: number, ContactAt: number?, Loop: boolean?, Path: { {number} }? }   (Path rows { t, x, z }, HRP space, studs)
	StartAt(Clip, Lead)                where a kick starts so the foot meets the ball Lead s later (0 with no contact)
	Hold(Clip, Lead, Margin, MaxHold)  HoldAt, Creep: a charge plays 1x to HoldAt, then Creep reaches StartAt at MaxHold
	Window(Clip)                       t0, t1: where the path's distance passes 10 % and 90 % of its peak
	Fit(Clip, DashTime, Cap)           Rate, Time, t0, t1: the clip's rate through its window (1..Cap) and the dash time
	Progress(Clip, t, t0, t1)          0..1 along the window's displacement by clip time t (clamped, never decreasing)
	KickSlot(Speed, RunSpeed)          "KickRun" at or above RunSpeed, else "Kick"
	DiveSlot(RightDot)                 "KeeperDiveRight" when the dive goes right (RightDot >= 0), else "KeeperDiveLeft"
	Spec: docs/superpowers/specs/2026-11-24-soccer-2-0-design.md §3-4. ]]
local A = {}

export type Clip = { Length: number, ContactAt: number?, Loop: boolean?, Path: { { number } }? }

local function at(P: { { number } }, t: number): (number, number)
	if t <= P[1][1] then return P[1][2], P[1][3] end
	for I = 2, #P do
		local B = P[I]
		if t <= B[1] then
			local Q = P[I - 1]
			local u = (t - Q[1]) / math.max(B[1] - Q[1], 1e-6)
			return Q[2] + (B[2] - Q[2]) * u, Q[3] + (B[3] - Q[3]) * u
		end
	end
	local L = P[#P]
	return L[2], L[3]
end

local function linear(t: number, t0: number, t1: number): number
	return math.clamp((t - t0) / math.max(t1 - t0, 1e-3), 0, 1)
end

function A.StartAt(Clip: Clip, Lead: number): number
	local C = Clip.ContactAt
	if not C then return 0 end
	return math.max(0, C - Lead)
end

function A.Hold(Clip: Clip, Lead: number, Margin: number, MaxHold: number): (number, number)
	local Start = A.StartAt(Clip, Lead)
	local HoldAt = math.min(math.max(0, (Clip.ContactAt or 0) - Margin), Start)
	return HoldAt, (Start - HoldAt) / math.max(MaxHold, 1e-3)
end

function A.Window(Clip: Clip): (number, number)
	local P = Clip.Path
	if not P or #P < 2 then return 0, Clip.Length end
	local Peak = 0
	for _, S in P do Peak = math.max(Peak, math.sqrt(S[2] * S[2] + S[3] * S[3])) end
	if Peak < 1e-3 then return 0, Clip.Length end
	local t0: number?, t1: number? = nil, nil
	for _, S in P do
		local M = math.sqrt(S[2] * S[2] + S[3] * S[3])
		if t0 == nil and M >= 0.1 * Peak then t0 = S[1] end
		if t1 == nil and M >= 0.9 * Peak then t1 = S[1] end
	end
	return t0 or 0, t1 or Clip.Length
end

function A.Fit(Clip: Clip, DashTime: number, Cap: number): (number, number, number, number)
	local t0, t1 = A.Window(Clip)
	local W = math.max(t1 - t0, 1e-3)
	local Rate = math.clamp(W / math.max(DashTime, 1e-3), 1, Cap)
	return Rate, W / Rate, t0, t1
end

function A.Progress(Clip: Clip, t: number, t0: number, t1: number): number
	local P = Clip.Path
	if not P or #P < 2 then return linear(t, t0, t1) end
	local x0, z0 = at(P, t0)
	local x1, z1 = at(P, t1)
	local Dx, Dz = x1 - x0, z1 - z0
	local Len2 = Dx * Dx + Dz * Dz
	if Len2 < 1e-6 then return linear(t, t0, t1) end
	-- the furthest the path has got along the window's direction by t (30 fps steps), so it never goes back
	local Stop = math.clamp(t, t0, t1)
	local Best, T = 0, t0
	while true do
		local x, z = at(P, T)
		Best = math.max(Best, ((x - x0) * Dx + (z - z0) * Dz) / Len2)
		if T >= Stop then break end
		T = math.min(T + 1 / 30, Stop)
	end
	return math.clamp(Best, 0, 1)
end

function A.KickSlot(Speed: number, RunSpeed: number): string
	return if Speed >= RunSpeed then "KickRun" else "Kick"
end

function A.DiveSlot(RightDot: number): string
	return if RightDot >= 0 then "KeeperDiveRight" else "KeeperDiveLeft"
end

return A
```

```lua
--[[
	Soccer clips (client, soccer-2.0): which animation a Config.Sports.SoccerAnims slot plays, its timing data
	(Config.SoccerClips, written by tools/soccer2/bake/bake_soccer_clips.lua), and the dash that follows a clip's travel.
		Id(Slot) -> (id?, fier)   a published Id; in Studio the baked NHSoccerClips copy (registered); else Was; else nil.
		                          fier = the FIER clip plays (its timing data applies)
		Clip(Slot) -> data?       the slot's Config.SoccerClips.Clips entry
		Drive(Track, Clip, DashTime, Cap) -> (Rate, Time, t0, t1)   the clip through its travel window at the fitted rate
		DashPath(Root, Track, Clip, Dir, Dist, Rate, t0, t1, Time)  the mover's dash: Dist studs along Dir on the clip's curve
		Cancel()                  stops a running DashPath (a new dash takes over)
	Spec: docs/superpowers/specs/2026-11-24-soccer-2-0-design.md §2, §4.
]]
local ReplicatedStorage = game:GetService(`ReplicatedStorage`)
local RunService = game:GetService(`RunService`)
local KeyframeSequenceProvider = game:GetService(`KeyframeSequenceProvider`)

local Config = require(ReplicatedStorage:WaitForChild(`Config`):WaitForChild(`Sports`))
local Anim = require(ReplicatedStorage:WaitForChild(`Functions`):WaitForChild(`SoccerAnim`))

local M = {}
local Data = nil :: any
local Temp = {} :: { [string]: string }
local Warned = {} :: { [string]: boolean }
local Token = 0

function M.Data(): any?
	if Data == nil then
		local Mod = ReplicatedStorage:FindFirstChild(`Config`) and ReplicatedStorage.Config:FindFirstChild(`SoccerClips`)
		local Ok, D = false, nil
		if Mod then Ok, D = pcall(require, Mod) end
		Data = if Ok and type(D) == `table` and type(D.Clips) == `table` then D else false
	end
	return Data or nil
end

function M.Clip(Slot: string): any?
	local E = Config.SoccerAnims[Slot]
	local D = M.Data()
	return if type(E) == `table` and E.Clip and D then D.Clips[E.Clip] else nil
end

function M.Id(Slot: string): (string?, boolean)
	local E = Config.SoccerAnims[Slot]
	if type(E) == `number` then return `rbxassetid://{E}`, false end
	if type(E) ~= `table` then return nil, false end
	if E.Id then return `rbxassetid://{E.Id}`, E.Clip ~= nil end
	if E.Clip and RunService:IsStudio() then
		local Got = Temp[E.Clip]
		if not Got then
			local F = ReplicatedStorage:FindFirstChild(`NHSoccerClips`)
			local KS = F and F:FindFirstChild(E.Clip)
			if KS and KS:IsA(`KeyframeSequence`) then
				local Ok, T = pcall(function() return KeyframeSequenceProvider:RegisterKeyframeSequence(KS) end)
				if Ok and T then Got = T Temp[E.Clip] = T end
			end
		end
		if Got then return Got, true end
		if not Warned[E.Clip] then
			Warned[E.Clip] = true
			warn(`[Soccer clips] {E.Clip} isn't in ReplicatedStorage.NHSoccerClips (run tools/soccer2/bake/bake_soccer_clips.lua)`)
		end
	end
	if E.Was then return `rbxassetid://{E.Was}`, false end
	return nil, false
end

function M.Drive(Track: AnimationTrack, Clip: any, DashTime: number, Cap: number): (number, number, number, number)
	local Rate, Time, t0, t1 = Anim.Fit(Clip, DashTime, Cap)
	Track.TimePosition = t0
	Track:AdjustSpeed(Rate)
	task.spawn(function()
		local T0 = os.clock()
		while Track.IsPlaying and Track.TimePosition < t1 and os.clock() - T0 < Time + 1 do RunService.Heartbeat:Wait() end
		if Track.IsPlaying and math.abs(Track.Speed - Rate) < 1e-3 then Track:AdjustSpeed(1) end
	end)
	return Rate, Time, t0, t1
end

function M.Cancel() Token += 1 end

function M.DashPath(Root: BasePart, Track: AnimationTrack, Clip: any, Dir: Vector3?, Dist: number, Rate: number, t0: number, t1: number, Time: number)
	Token += 1
	local Mine = Token
	local D = if typeof(Dir) == `Vector3` then Vector3.new(Dir.X, 0, Dir.Z) else Vector3.zero
	if D.Magnitude < 0.1 then D = Vector3.new(Root.CFrame.LookVector.X, 0, Root.CFrame.LookVector.Z) end
	D = D.Unit
	local Att = Instance.new(`Attachment`) Att.Parent = Root
	local LV = Instance.new(`LinearVelocity`)
	LV.Attachment0 = Att LV.RelativeTo = Enum.ActuatorRelativeTo.World LV.VelocityConstraintMode = Enum.VelocityConstraintMode.Plane
	LV.PrimaryTangentAxis = Vector3.new(1, 0, 0) LV.SecondaryTangentAxis = Vector3.new(0, 0, 1)
	LV.PlaneVelocity = Vector2.zero
	LV.MaxForce = 60000
	LV.Parent = Root
	local T0 = os.clock()
	local Conn: RBXScriptConnection? = nil
	local function finish() if Conn then Conn:Disconnect() end LV:Destroy() Att:Destroy() end
	Conn = RunService.Heartbeat:Connect(function()
		local t = Track.TimePosition
		if Mine ~= Token or not Track.IsPlaying or t >= t1 or not Root.Parent or os.clock() - T0 > Time + 0.5 then finish() return end
		local t2 = math.min(t + 1 / 30, t1)
		local V = if t2 - t > 1e-4 then Dist * (Anim.Progress(Clip, t2, t0, t1) - Anim.Progress(Clip, t, t0, t1)) / (t2 - t) * Rate else 0
		LV.PlaneVelocity = Vector2.new(D.X, D.Z) * V
	end)
end

return M
```

```lua
-- a filled slot plays its clip (the FIER clip, or the 1.1 one); returns the track, or nil for the procedural pose.
-- Opts: Slot (instead of SLOT[Kind]), At (TimePosition), Speed.
function Local.Play(Char: Model, Kind: string, Opts: { Slot: string?, At: number?, Speed: number? }?): AnimationTrack?
	local Slot = (Opts and Opts.Slot) or SLOT[Kind] or ``
	local Id, Fier = Clips.Id(Slot)
	if not Id then return nil end
	local Hum = Char:FindFirstChildOfClass(`Humanoid`)
	local Animator = Hum and Hum:FindFirstChildOfClass(`Animator`)
	if not Animator then return nil end
	Tracks[Char] = Tracks[Char] or {}
	local Key = Slot .. (if Fier then `` else `~`)
	local Tr = Tracks[Char][Key]
	if not Tr then
		local A = Instance.new(`Animation`)
		A.AnimationId = Id
		local Ok, Loaded = pcall(Animator.LoadAnimation, Animator, A)
		if not Ok then return nil end
		Tr = Loaded
		Tr.Priority = Enum.AnimationPriority.Action2
		local C = Fier and Clips.Clip(Slot)
		Tr.Looped = C ~= nil and C.Loop == true   -- 1.1 clips (the slide) are saved looping
		Tracks[Char][Key] = Tr
	end
	-- one soccer clip at a time per body
	for K, Other in Tracks[Char] do if K ~= Key and Other.IsPlaying then Other:Stop(0.08) end end
	Tr:Play(0.08)
	if Opts and Opts.At then Tr.TimePosition = Opts.At end
	Tr:AdjustSpeed(Opts and Opts.Speed or 1)
	-- FIER clips play out; 1.1 clips are cut to the move; long clips are cut to the moment
	local Cut = STOP_AFTER[Kind] or (if Fier then nil else LENGTH[Kind])
	if Cut then task.delay(Cut, function() if Tr.IsPlaying then Tr:Stop(0.2) end end) end
	return Tr
end

function Local.Pose(Char: Model?, Kind: string, Side: number?, Opts: { Slot: string?, At: number?, Speed: number? }?): AnimationTrack?
	if not Char then return nil end
	local Tr = Local.Play(Char, Kind, Opts)
	if Tr then return Tr end
	Poses[Char] = { Kind = Kind; Start = os.clock(); Length = LENGTH[Kind] or 0.5; Side = Side }
	return nil
end

-- (soccer-2.0) the charge: 1x to MARGIN before contact, then a creep that reaches the kick point at MaxHold
local Charging = {} :: { [Model]: AnimationTrack }
function Local.Charge(Char: Model?)
	if not Char then return end
	local Tr = Local.Pose(Char, `Charge`)
	local C = Local.Data(`Kick`)
	if not (Tr and C) then return end
	Charging[Char] = Tr
	local HoldAt, Creep = Anim.Hold(C, LEAD, MARGIN, T.Kick.MaxHold)

		local T0 = os.clock()

		if not (Tr.IsPlaying and Charging[Char] == Tr) then return end

-- (soccer-2.0) a kick from the server: the clip jumps to LEAD s before contact (or keeps the charge's place if it's ahead)
function Local.Kick(Char: Model?, Kind: string)

		local R = Char:FindFirstChild(`HumanoidRootPart`) :: BasePart?

		Slot = Anim.KickSlot(Vector3.new(V.X, 0, V.Z).Magnitude, RUN_SPEED)
	end

	end

-- (soccer-2.0) a move with travel: every client plays the clip through its window at the fitted rate; the mover's own
-- client also runs the dash (the 1.1 distance and direction on the clip's speed curve). No clip data: the 1.1 dash.
function Local.Move(Char: Model?, Kind: string, Slot: string, Side: number?, Me: boolean, Dir: Vector3?, Dist: number, Time: number)
	if not Char then return end
	local C = Local.Data(Slot)
	local Tr = Local.Pose(Char, Kind, Side, { Slot = Slot })
```

```lua
local Clips = require(Controllers:WaitForChild(`Gui`):WaitForChild(`Soccer`):WaitForChild(`Clips`))

local Shared = {}
local Local = {}

local On = false
local IdleSlot = nil :: string?
local Keeper = false
local SportsMod = nil :: any

local function ac(): any?
	local Ok, AC = pcall(require, script.Parent:WaitForChild(`AnimationController`))
	return if Ok then AC else nil
end
local function sports(): any?
	if SportsMod == nil then
		local Ok, S = pcall(require, Controllers.Gui:WaitForChild(`Sports`))
		SportsMod = if Ok then S else false
	end
	return SportsMod or nil
end
local function anim(Name: string, Id: string): Animation
	local A = Instance.new(`Animation`)
	A.Name = Name
	A.AnimationId = Id
	return A
end

function Local.IsSoccer(): boolean
	local V = Player:GetAttribute(`InSport`)
	if type(V) ~= `string` then return false end
	local Root = workspace:FindFirstChild(`Sports`)
	local Venue = Root and Root:FindFirstChild(V)
	return Venue ~= nil and Venue:GetAttribute(`SportVenue`) == `Soccer`
end

function Local.IsKeeper(): boolean
	local S = sports()
	local St = S and S.State()
	return St ~= nil and St.Keeper == true and St.InZone == true
end

function Local.WantIdle(): string
	if Local.IsKeeper() then return `KeeperIdle` end
	local Char = Player.Character
	if Char and Char:GetAttribute(`HasSportBall`) then return `IdleBall` end
	return `Idle`
end

function Local.Pack(): Folder?
	local Idle, Jog = Clips.Id(`Idle`), Clips.Id(`Jog`)
	if not (Idle and Jog) then return nil end
	local F = Instance.new(`Folder`)
	F.Name = `SoccerMovement`
	for _, A in ReplicatedStorage.Assets.Animations.Default.Movement:GetChildren() do
		if A:IsA(`Animation`) and A.Name ~= `Idle` and A.Name ~= `Walk` then A:Clone().Parent = F end
	end
	anim(`Idle`, Idle).Parent = F
	anim(`Walk`, Jog).Parent = F
	return F
end

function Local.Override()
	local AC = ac()
	if not AC then return end
	local D = Clips.Data()
	local St = (D and D.Stride) or {}
	local O = { Stride = { Walk = St.SoccerJog; Back = St.SoccerJogBack }; Back = (Clips.Id(`JogBack`)) }
	Keeper = Local.IsKeeper()
	if Keeper then
		O.Left = (Clips.Id(`KeeperStepLeft`))
		O.Right = (Clips.Id(`KeeperStepRight`))
		O.Stride.StrafeLeft = St.KeeperStep
		O.Stride.StrafeRight = St.KeeperStep
	end
	AC.Override(O)
end

-- swap the Idle track in place (no restart of a playing walk)
function Local.SetIdle(Slot: string)
	local AC = ac()
	local AM = AC and AC.Controller
	if not (AM and AM.Animator and AM.TrackTable and AM.TrackTable.Idle) then return end
	local Id = Clips.Id(Slot) or Clips.Id(`Idle`)
	if not Id then return end
	IdleSlot = Slot
	pcall(function()
		local Old = AM.TrackTable.Idle
		local Playing = AM.Pose == `Idle` and Old.IsPlaying
		local New = AM.Animator:LoadAnimation(anim(`Idle`, Id))
		New.Priority = Enum.AnimationPriority.Idle
		New.Looped = true
		AM.TrackTable.Idle = New
		Old:Stop(0.2)
		task.delay(0.3, function() pcall(function() Old:Destroy() end) end)
		if Playing then New:Play(0.2) end
	end)
end

function Local.Apply()
	local AC = ac()
	local AM = AC and AC.Controller
	local Char = Player.Character
	-- the AnimationController must already be on this body (it rebuilds the pack a moment after a respawn)
	if not (AM and AM.Animator and Char and AM.Animator:IsDescendantOf(Char) and AM.TrackTable.Idle) then return end
	local Pack = Local.Pack()
	if not Pack then return end
	On = true
	Local.Override()
	pcall(function()
		AM:StopAllTracks():SetTracks(Pack):PlayTrack(`Idle`, 0.2)
		for _, N in { `Idle`, `Walk` } do local T = AM.TrackTable[N] if T then T.Looped = true end end
	end)
	IdleSlot = `Idle`
	local Want = Local.WantIdle()
	if Want ~= IdleSlot then Local.SetIdle(Want) end
end

function Local.Restore()
	On = false
	IdleSlot = nil
	Keeper = false
	local AC = ac()
	if not AC then return end
	AC.Override(nil)
	local AM = AC.Controller
	local Char = Player.Character
	if not (AM and AM.Animator and Char) then return end
	-- the same choice as Transformation's restorePack: the weapon pack when equipped, else Default
	local Hum = Char:FindFirstChildWhichIsA(`Humanoid`)
	local W = Char:GetAttribute(`Weapon`)
	local Pack = (Hum and Hum:GetAttribute(`Equip`) and W and ReplicatedStorage.Assets.Animations.Weapons:FindFirstChild(W)) or ReplicatedStorage.Assets.Animations.Default
	pcall(function() AM:StopAllTracks():SetTracks(Pack.Movement):PlayTrack(`Idle`, 0.15) end)
end

function Local.Step()
	local Want = Local.IsSoccer()
	if Want and not On then Local.Apply() elseif not Want and On then Local.Restore() end
	if not On then return end
	if Local.IsKeeper() ~= Keeper then Local.Override() end
	local Idle = Local.WantIdle()
	if Idle ~= IdleSlot then Local.SetIdle(Idle) end
end

function Shared.OnStart()
	-- a light poll (4 Hz): InSport, the keeper's zone and the ball come from three places
	task.spawn(function()
		while true do
			local Ok, Err = pcall(Local.Step)
			if not Ok then warn(`[SoccerPack] {Err}`) end
			task.wait(0.25)
		end
	end)
	-- a new body starts on the default pack; the next tick re-applies (Apply waits for the controller to be on it)
	Player.CharacterAdded:Connect(function() On = false IdleSlot = nil end)
end

-- (soccer-2.0) Studio checks
function Shared.StateForTests() return { On = On; Idle = IdleSlot; Keeper = Keeper } end

return Shared
```

```lua
-- soccer_demo (soccer-2.0 Task 6, Client, Play, during a solo match): about 9 s of jog, charge, kick, dribble, backpedal
-- and dive through the real handlers, while it dumps every rig part per frame into _G.NHSoccerDump (JSON in the audit
-- format: parts "H:<Part>" rows [cx, cy, cz, tipx, tipy, tipz, lowest, 0, 0, 0] with z up and the floor at 0; cam rows).
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local Centre = workspace.Sports.Pitch:GetBoundingBox().Position
local D = Vector3.new(Centre.X - Root.Position.X, 0, Centre.Z - Root.Position.Z)
D = if D.Magnitude > 1 then D.Unit else Vector3.new(Root.CFrame.LookVector.X, 0, Root.CFrame.LookVector.Z).Unit
local RP = RaycastParams.new() RP.FilterType = Enum.RaycastFilterType.Exclude RP.FilterDescendantsInstances = { Char }

for _, N in PARTS do Rows["H:" .. N] = {} Rows["V:" .. N] = {} end
local function r(v) return math.floor(v * 1000 + 0.5) / 1000 end
local function lowest(Pt)
	local Lo = math.huge
	for _, X in { -0.5, 0.5 } do for _, Y in { -0.5, 0.5 } do for _, Z in { -0.5, 0.5 } do
		Lo = math.min(Lo, (Pt.CFrame * (Pt.Size * Vector3.new(X, Y, Z))).Y)

			local C = Pt.Position

			local Row = { r(C.X), r(C.Z), r(C.Y - Floor), r(Tip.X), r(Tip.Z), r(Tip.Y - Floor), r(lowest(Pt)), 0, 0, 0 }
			table.insert(Rows["H:" .. N], Row)
			table.insert(Rows["V:" .. N], Row)
		end
	end
	local CF = workspace.CurrentCamera.CFrame
	table.insert(Cam, { "C", r(CF.X), r(CF.Z), r(CF.Y), 1, 0, 0, 0 })
end)
local Move = Vector3.zero

task.wait(0.5)                                   -- idle (Soccer Idle)
Move = D task.wait(1.8)                          -- jog (Jog Forward, stride-matched)
Move = Vector3.zero task.wait(0.4)
Soccer.FxForTests("S_Charge", { Player = P }) task.wait(0.7)   -- the creeping wind-up
Soccer.FxForTests("S_Kick", { Player = P; Kind = "Shot"; Power = 0.8 }) task.wait(0.9)
```

```python
"""Pull _G.NHSoccerDump (written by checks/soccer_demo.lua) from the Play client in 20,000-character chunks.
usage: python -X utf8 -I tools/soccer2/harness/pull_dump.py <out.json>"""
import os, subprocess, sys
HERE = os.path.dirname(os.path.abspath(__file__))
RUN = os.path.join(os.path.dirname(os.path.dirname(HERE)), "admin-console", "run_live.py")
TMP = os.path.join(HERE, "_pull.lua")
out, i = [], 0
while True:
    open(TMP, "w", encoding="utf-8").write('local S = _G.NHSoccerDump\nif not S then return "FAIL no dump" end\n'
        'local a = %d * 20000 + 1\nif a > #S then return "PASS <<EOF>>" end\nreturn "PASS " .. S:sub(a, a + 19999)' % i)
    r = subprocess.run([sys.executable, "-X", "utf8", "-I", RUN, TMP, "30", "Client"], capture_output=True, text=True, encoding="utf-8")
    s = r.stdout.rstrip("\r\n")
    if not s.startswith("PASS "): sys.exit("pull failed: " + s[:300])
    s = s[5:]
    if s == "<<EOF>>": break
    out.append(s)
    i += 1
open(sys.argv[1], "w", encoding="utf-8").write("".join(out))
print("pulled", i, "chunks,", sum(len(x) for x in out), "chars")
```
